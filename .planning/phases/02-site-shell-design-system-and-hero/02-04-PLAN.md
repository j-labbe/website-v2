           background-color: var(--color-bg);
           line-height: 1.6;
           min-height: 100vh;
           position: relative;
         }

         img, svg {
           display: block;
           max-width: 100%;
         }

         a {
           color: inherit;
           text-decoration: none;
         }

           font: inherit;
           color: inherit;
           background: none;
           border: none;
           cursor: pointer;
         }

           position: fixed;
           inset: 0;
           pointer-events: none;
           z-index: 0;

           background-size: 40px 40px;
           transition: opacity 250ms ease;
         }
       }

       /* Component layer: reusable animation classes */
       @layer components {
         .skeleton {
           background: linear-gradient(
             100deg,
             var(--color-surface) 40%,
             var(--color-surface-2) 50%,
             var(--color-surface) 60%

           animation: shimmer 1.5s ease-in-out infinite;
           border-radius: 8px;

           animation: fadeInUp 400ms ease-out forwards;
           animation-delay: calc(var(--stagger-index, 0) * 100ms);
         }
       }

       /* Reduced motion */
       @media (prefers-reduced-motion: reduce) {
         .skeleton {
           animation: none;
         }

         .stagger-item {
           animation: none;
           opacity: 1;
           transform: none;
         }
       }
